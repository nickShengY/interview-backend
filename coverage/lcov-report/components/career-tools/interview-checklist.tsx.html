  const toggleItem = (id: string) => {
    const updated = items.map(i => i.id === id ? { ...i, checked: !i.checked } : i)
    setItems(updated)
    saveChecklist(updated)
  }
 
  const resetAll = () => {
    const reset = items.map(i => ({ ...i, checked: false }))
